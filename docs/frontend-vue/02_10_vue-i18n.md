# Internationalization

Internationalizing Vue.js applications in Tauri environments requires special considerations for loading locale data and managing translations. This article covers comprehensive i18n strategies for desktop applications.

## Setting Up Vue I18n

### Installation

```bash
npm install vue-i18n@9
```

### Basic Configuration

```javascript
// src/i18n/index.js
import { createI18n } from 'vue-i18n'
import { invoke } from '@tauri-apps/api/tauri'

// Load locale messages dynamically from backend
async function loadLocaleMessages(locale) {
  try {
    // Load from backend command
    return await invoke('get_locale_messages', { locale })
  } catch (error) {
    console.warn(`Failed to load locale ${locale}:`, error)
    // Fallback to English
    return await invoke('get_locale_messages', { locale: 'en' })
  }
}

// Create initial i18n instance with English as fallback
const i18n = createI18n({
  legacy: false, // Use Composition API
  locale: 'en',
  fallbackLocale: 'en',
  messages: {}
})

export default i18n
```

### Initialize i18n with System Locale

```javascript
// src/main.js
import { createApp } from 'vue'
import App from './App.vue'
import i18n from './i18n'

const initApp = async () => {
  const app = createApp(App)
  
  // Get system locale
  try {
    const systemLocale = await invoke('get_system_locale')
    i18n.global.locale.value = systemLocale || 'en'
  } catch (error) {
    console.warn('Failed to get system locale, using default:', error)
  }
  
  // Load initial locale messages
  const messages = await loadLocaleMessages(i18n.global.locale.value)
  i18n.global.setLocaleMessage(i18n.global.locale.value, messages)
  
  app.use(i18n)
  app.mount('#app')
}

initApp()
```

## Dynamic Locale Switching

### Locale Manager Composable

```javascript
// composables/useLocaleManager.js
import { ref, computed } from 'vue'
import { invoke } from '@tauri-apps/api/tauri'

export function useLocaleManager() {
  const currentLocale = ref('en')
  const availableLocales = ref([])
  const loading = ref(false)
  const error = ref(null)

  const localeNames = {
    'en': 'English',
    'es': 'Español', 
    'fr': 'Français',
    'de': 'Deutsch',
    'ja': '日本語',
    'zh': '中文'
  }

  const loadAvailableLocales = async () => {
    try {
      availableLocales.value = await invoke('get_available_locales')
    } catch (err) {
      error.value = err.message
    }
  }

  const changeLocale = async (locale) => {
    if (!availableLocales.value.includes(locale)) {
      throw new Error(`Locale ${locale} is not available`)
    }

    loading.value = true
    error.value = null

    try {
      const messages = await invoke('get_locale_messages', { locale })
      
      // Update Vue I18n instance
      const { global } = getCurrentInstance().appContext.config.globalProperties.$i18n
      global.setLocaleMessage(locale, messages)
      global.locale.value = locale
      
      currentLocale.value = locale
      
      // Save user preference
      await invoke('save_user_locale', { locale })
    } catch (err) {
      error.value = err.message
      throw err
    } finally {
      loading.value = false
    }
  }

  const getLocaleName = (locale) => {
    return localeNames[locale] || locale
  }

  const isRTL = computed(() => {
    // Right-to-left languages
    const rtlLocales = ['ar', 'he', 'fa', 'ur']
    return rtlLocales.includes(currentLocale.value)
  })

  return {
    currentLocale,
    availableLocales,
    loading,
    error,
    localeNames,
    loadAvailableLocales,
    changeLocale,
    getLocaleName,
    isRTL
  }
}
```

### Locale-Specific Number and Date Formatting

```javascript
// composables/useLocaleFormatting.js
import { computed } from 'vue'
import { useI18n } from 'vue-i18n'

export function useLocaleFormatting() {
  const { locale } = useI18n()
  
  const formatNumber = (value, options = {}) => {
    return new Intl.NumberFormat(locale.value, {
      style: 'decimal',
      minimumFractionDigits: 0,
      maximumFractionDigits: 2,
      ...options
    }).format(value)
  }
  
  const formatCurrency = (value, currency = 'USD') => {
    return new Intl.NumberFormat(locale.value, {
      style: 'currency',
      currency
    }).format(value)
  }
  
  const formatDate = (date, options = {}) => {
    return new Intl.DateTimeFormat(locale.value, {
      year: 'numeric',
      month: 'long',
      day: 'numeric',
      ...options
    }).format(date)
  }
  
  const formatRelativeTime = (date) => {
    const rtf = new Intl.RelativeTimeFormat(locale.value, { numeric: 'auto' })
    const diffInSeconds = (date - new Date()) / 1000
    const diffInMinutes = diffInSeconds / 60
    const diffInHours = diffInMinutes / 60
    const diffInDays = diffInHours / 24
    
    if (Math.abs(diffInDays) >= 1) {
      return rtf.format(Math.round(diffInDays), 'day')
    } else if (Math.abs(diffInHours) >= 1) {
      return rtf.format(Math.round(diffInHours), 'hour')
    } else {
      return rtf.format(Math.round(diffInMinutes), 'minute')
    }
  }
  
  return {
    formatNumber,
    formatCurrency,
    formatDate,
    formatRelativeTime
  }
}
```

## File-Based Locale Management

### Loading Locales from Files

```javascript
// utils/localeLoader.js
import { invoke } from '@tauri-apps/api/tauri'

export class LocaleLoader {
  constructor() {
    this.cache = new Map()
  }
  
  async loadLocaleFromBackend(locale) {
    if (this.cache.has(locale)) {
      return this.cache.get(locale)
    }
    
    try {
      const messages = await invoke('get_locale_messages', { locale })
      this.cache.set(locale, messages)
      return messages
    } catch (error) {
      console.error(`Failed to load locale ${locale}:`, error)
      throw error
    }
  }
  
  async loadLocaleFromFiles(locale) {
    try {
      // Load from JSON file in app resources
      const localeContent = await invoke('read_locale_file', { 
        filename: `locales/${locale}.json` 
      })
      
      const messages = JSON.parse(localeContent)
      this.cache.set(locale, messages)
      return messages
    } catch (error) {
      console.error(`Failed to load locale file ${locale}:`, error)
      throw error
    }
  }
  
  getAvailableLocales() {
    // Get available locales from a config file
    return invoke('get_available_locales_config')
  }
  
  clearCache() {
    this.cache.clear()
  }
}

export const localeLoader = new LocaleLoader()
```

### Locale Store

```javascript
// stores/locale.js
import { defineStore } from 'pinia'
import { localeLoader } from '@/utils/localeLoader'

export const useLocaleStore = defineStore('locale', {
  state: () => ({
    current: 'en',
    messages: {},
    available: [],
    loading: false,
    error: null
  }),
  
  actions: {
    async initialize() {
      await this.loadAvailableLocales()
      await this.detectSystemLocale()
    },
    
    async loadAvailableLocales() {
      try {
        this.available = await localeLoader.getAvailableLocales()
      } catch (error) {
        this.error = error.message
      }
    },
    
    async detectSystemLocale() {
      try {
        // Get system locale preference
        const systemLocale = await invoke('get_system_locale')
        if (this.available.includes(systemLocale)) {
          await this.setLocale(systemLocale)
        } else {
          await this.setLocale('en') // fallback
        }
      } catch (error) {
        await this.setLocale('en')
      }
    },
    
    async setLocale(locale) {
      if (!this.available.includes(locale)) {
        throw new Error(`Locale ${locale} is not available`)
      }
      
      this.loading = true
      this.error = null
      
      try {
        const messages = await localeLoader.loadLocaleFromBackend(locale)
        this.messages = messages
        this.current = locale
        
        // Update global i18n instance
        const { global } = this.$nuxt?.$i18n || this.$i18n
        if (global) {
          global.locale.value = locale
          global.setLocaleMessage(locale, messages)
        }
      } catch (error) {
        this.error = error.message
        throw error
      } finally {
        this.loading = false
      }
    }
  },
  
  getters: {
    currentLocaleName: (state) => {
      const names = {
        'en': 'English',
        'es': 'Español',
        'fr': 'Français',
        'de': 'Deutsch'
      }
      return names[state.current] || state.current
    },
    
    isRTL: (state) => {
      const rtlLocales = ['ar', 'he', 'fa', 'ur']
      return rtlLocales.includes(state.current)
    }
  }
})
```

## Backend Locale Commands

### Locale Detection and Management

```rust
// src-tauri/src/locales.rs (additional functions)

use std::env;
use sys_locale::get_locale;

#[tauri::command]
pub fn get_system_locale() -> Result<String, String> {
    match get_locale() {
        Some(locale) => {
            // Extract language code (e.g., "en-US" -> "en")
            let lang_code = locale.split('-').next().unwrap_or("en");
            Ok(lang_code.to_string())
        },
        None => Ok("en".to_string())
    }
}

#[tauri::command]
pub fn save_user_locale(locale: String) -> Result<(), String> {
    // Save user preference to config file
    let config_path = get_app_config_dir()
        .map_err(|e| format!("Failed to get config dir: {}", e))?
        .join("locale.json");
    
    let locale_json = serde_json::json!({
        "locale": locale,
        "timestamp": chrono::Utc::now().to_rfc3339()
    });
    
    std::fs::write(&config_path, locale_json.to_string())
        .map_err(|e| format!("Failed to save locale: {}", e))?;
    
    Ok(())
}

#[tauri::command]
pub fn get_user_preferred_locale() -> Result<String, String> {
    let config_path = get_app_config_dir()
        .map_err(|e| format!("Failed to get config dir: {}", e))?
        .join("locale.json");
    
    if config_path.exists() {
        let content = std::fs::read_to_string(&config_path)
            .map_err(|e| format!("Failed to read config: {}", e))?;
        
        let json: serde_json::Value = serde_json::from_str(&content)
            .map_err(|e| format!("Failed to parse config: {}", e))?;
        
        if let Some(locale) = json["locale"].as_str() {
            return Ok(locale.to_string());
        }
    }
    
    // Fallback to system locale
    get_system_locale()
}

fn get_app_config_dir() -> Result<std::path::PathBuf, String> {
    let config_dir = dirs::config_dir()
        .ok_or("Failed to get config directory")?
        .join("your-app-name");
    
    std::fs::create_dir_all(&config_dir)
        .map_err(|e| format!("Failed to create config dir: {}", e))?;
    
    Ok(config_dir)
}
```

## Performance Optimization

### Lazy Loading Locales

```javascript
// utils/lazyLocaleLoader.js
export class LazyLocaleLoader {
  constructor() {
    this.loadedLocales = new Set()
    this.loadingPromises = new Map()
  }
  
  async loadLocale(locale) {
    if (this.loadedLocales.has(locale)) {
      return this.getStoredMessages(locale)
    }
    
    // Prevent duplicate loading
    if (this.loadingPromises.has(locale)) {
      return this.loadingPromises.get(locale)
    }
    
    const loadPromise = this._loadLocaleInternal(locale)
    this.loadingPromises.set(locale, loadPromise)
    
    try {
      const messages = await loadPromise
      this.loadedLocales.add(locale)
      return messages
    } finally {
      this.loadingPromises.delete(locale)
    }
  }
  
  async _loadLocaleInternal(locale) {
    // Load locale from backend
    return await invoke('get_locale_messages', { locale })
  }
  
  async preloadCommonLocales() {
    const commonLocales = ['en', 'es', 'fr', 'de']
    const promises = commonLocales.map(locale => this.loadLocale(locale))
    await Promise.all(promises)
  }
}

export const lazyLocaleLoader = new LazyLocaleLoader()
```
